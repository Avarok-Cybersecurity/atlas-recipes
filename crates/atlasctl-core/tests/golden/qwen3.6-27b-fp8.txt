docker run -d --entrypoint '' --gpus all --ipc=host --shm-size=32gb --network=host --user 1000:1000 -v /etc/passwd:/etc/passwd:ro -v /etc/group:/etc/group:ro --security-opt no-new-privileges --security-opt seccomp=unconfined --cap-add=IPC_LOCK --cap-add=SYS_NICE --ulimit memlock=-1:-1 --ulimit stack=67108864 --device /dev/infiniband --label io.atlasctl.managed=1 --label io.atlasctl.recipe=qwen3.6-27b-fp8 --rm --name atlas-qwen3.6-27b-fp8 -e HF_HOME=/cache/huggingface -e HF_HUB_OFFLINE=1 -e TRANSFORMERS_OFFLINE=1 -v /home/spark/.cache/huggingface:/cache/huggingface avarok/atlas-gb10:latest spark serve Qwen/Qwen3.6-27B-FP8 --port 8888 --host 0.0.0.0 --gpu-memory-utilization 0.88 --max-seq-len 262144 --max-num-seqs 2 --kv-cache-dtype fp8 --max-batch-size 2 --kv-high-precision-layers auto --scheduling-policy slai --oom-guard-mb 1024 --mtp-quantization bf16 --enable-prefix-caching
